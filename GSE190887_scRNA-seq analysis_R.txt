Step 49: Filter cell types and summarize Il33 expression
library(dplyr)
library(ggplot2)

il33_cells <- c("Fib", "Myofib", "Uro")

seurat_il33 <- subset(
  seurat_obj,
  subset = celltype0421 %in% il33_cells
)

seurat_il33$celltype0421 <- factor(
  seurat_il33$celltype0421,
  levels = c("Fib", "Myofib", "Uro")
)

seurat_il33$model <- factor(
  seurat_il33$model,
  levels = c("Health", "IRI", "UUO")
)

il33_expr <- FetchData(
  seurat_il33,
  vars = il33_gene_ids_seurat[1]
)

colnames(il33_expr) <- "Il33"

il33_expr$celltype <- seurat_il33$celltype0421
il33_expr$model <- seurat_il33$model

il33_summary <- il33_expr %>%
  group_by(celltype, model) %>%
  summarise(
    avg_expression = mean(Il33),
    pct_expressed = mean(Il33 > 0) * 100,
    .groups = "drop"
  )

il33_summary
