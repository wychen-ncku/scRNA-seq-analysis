Step 40: Generate DotPlot of IL33 across conditions in target lineages
# IL33 expression in EC and VSMC/Pericyte/Fib
# Ref / AKI / CKD
# Custom ggplot2 DotPlot

# ------------------------------------------------------------
1. Select EC and VSMC/Pericyte/Fib
# ------------------------------------------------------------
seurat_IL33_EC_VSMC <- subset(
    seurat_obj,
    subset = celltype %in% c(
        "EC",
        "VSMC/Pericyte/Fib"
    )
)


# ------------------------------------------------------------
# 2. Create celltype × condition group
# ------------------------------------------------------------

seurat_IL33_EC_VSMC$celltype_condition <- paste(
    seurat_IL33_EC_VSMC$celltype,
    seurat_IL33_EC_VSMC$condition.l1,
    sep = "_"
)


# ------------------------------------------------------------
# 3. Generate DotPlot data
# ------------------------------------------------------------

dot_data <- DotPlot(
    seurat_IL33_EC_VSMC,
    features = "IL33",
    group.by = "celltype_condition"
)$data


# ------------------------------------------------------------
# 4. Extract condition and cell type
# ------------------------------------------------------------

dot_data$condition <- sub(
    ".*_(Ref|AKI|CKD)$",
    "\\1",
    dot_data$id
)

dot_data$celltype <- sub(
    "_(Ref|AKI|CKD)$",
    "",
    dot_data$id
)


# ------------------------------------------------------------
# 5. Set order
# ------------------------------------------------------------

dot_data$condition <- factor(
    dot_data$condition,
    levels = c(
        "Ref",
        "AKI",
        "CKD"
    )
)

dot_data$celltype <- factor(
    dot_data$celltype,
    levels = c(
        "EC",
        "VSMC/Pericyte/Fib"
    )
)


# ------------------------------------------------------------
# 6. Display labels
# ------------------------------------------------------------

dot_data$celltype_label <- factor(
    dot_data$celltype,
    levels = c(
        "VSMC/Pericyte/Fib",
        "EC"
    ),
    labels = c(
        "VSMC/Pericyte/Fib",
        "EC"
    )
)

# ------------------------------------------------------------
# 7. Draw plot
# ------------------------------------------------------------

p_IL33_EC_VSMC <- ggplot(
    dot_data,
    aes(
        x = condition,
        y = celltype_label
    )
) +
    
    geom_point(
        aes(
            size = pct.exp,
            color = avg.exp
        )
    ) +
    
    # Average Expression
    scale_color_gradientn(
        colors = c(
            "#D9D9D9",
            "#9B7FD3",

    # Percent Expressed

    # ========================================================
    # Average Expression = TOP
